### Problem Statement

After v1.33.0 deployment (35 patterns, compact TP/SL), three operational gaps remained:

1. **No OOS validation at scan time** — Scanner selected patterns using only IS data, no holdout check
2. **No staleness detection** — Old pattern files could run indefinitely without warning
3. **No drawdown-responsive sizing** — Fixed position sizing regardless of equity drawdown
4. **Trade data lost to log rotation** — Complete trade analysis impossible after log files rotate

```python
def get_position_size(self, state: dict) -> float:
    """
    동적 포지션 사이징 (MDD 기반)

    Args:
        state: BotState with peak_equity

    Returns:
        size_multiplier: 1.0 (정상) ~ 0.25 (DD 20%+)
    """
    if not config.get('mdd_sizing', {}).get('enabled', True):
        return 1.0

    peak_equity = state.get('peak_equity', current_equity)
    current_equity = get_account_balance()

    if peak_equity <= 0:
        return 1.0

    dd_pct = (peak_equity - current_equity) / peak_equity

    if dd_pct <= 0.05:
        return 1.0
    elif dd_pct >= 0.20:
        return 0.25
    else:
        # 선형 보간
        return 1.0 - (dd_pct - 0.05) / 0.15 * 0.75
```

```python
def _check_scan_staleness(self) -> None:
    """Check if dynamic_patterns.json is stale (>90 days)"""
    patterns_file = self.config_manager.get_dynamic_patterns_path()
    if not os.path.exists(patterns_file):
        return

    patterns_data = load_json(patterns_file)
    generated_at_str = patterns_data.get('metadata', {}).get('generated_at')

    if not generated_at_str:
        logger.warning("No generated_at timestamp in patterns file")
        return

    generated_at = datetime.fromisoformat(generated_at_str)
    days_old = (datetime.now() - generated_at).days

    rescan_interval = self.config.get('strategy', {}).get('rescan_interval_days', 90)

    if days_old > rescan_interval:
        logger.warning(
            f"Scan staleness: patterns are {days_old}d old (threshold: {rescan_interval}d). "
            f"Consider re-running scanner: pattern_scanner.py --discovery-method mae_mfe"
        )
```

#### 6. pattern_scanner.py (Holdout Validation)
```python
def holdout_validate(df, patterns, holdout_days=7):
    """
    Holdout OOS validation: last N days

    - Exclude last N days from IS
    - Backtest patterns on holdout period only
    - Remove patterns with WR Excess ≤ 0
    """
    n_bars = holdout_days * 288  # 5m bars per day (24h / 5m)
    is_df = df[:-n_bars]
    holdout_df = df[-n_bars:]

    holdout_results = []

    for pattern in patterns:
        # Backtest on holdout only
        trades = bt_signals(holdout_df, pattern)
        if len(trades) < MIN_TRADES:
            continue

        wr = sum(1 for t in trades if t['pnl'] > 0) / len(trades)
        rw_wr = pattern['tp'] / (pattern['tp'] + pattern['sl'])
        wr_excess = wr - rw_wr

        if wr_excess <= 0:
            logger.info(f"Holdout SKIP: {pattern['name']} WR Excess {wr_excess:.2f}pp")
            continue

        holdout_results.append({
            'name': pattern['name'],
            'holdout_wr': wr,
            'holdout_wr_excess': wr_excess,
            'holdout_trades': len(trades)
        })

    return holdout_results

# main()에서 호출
def main():
    # ... existing code ...

    # Holdout split
    patterns = scan_patterns(df, **kwargs)
    holdout_results = holdout_validate(df, patterns, holdout_days=7)

    # Holdout 검증 패턴만 유지
    validated_patterns = [p for p in patterns if any(h['name'] == p['name'] for h in holdout_results)]

    # 출력 JSON에 holdout 섹션 추가
    output = build_output_json(
        validated_patterns,
        holdout_results=holdout_results,
        generated_at=datetime.now().isoformat()
    )
```
